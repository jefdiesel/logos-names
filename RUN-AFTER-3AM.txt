run() { echo "\$ names $*" >> $LOG; $N "$@" >> $LOG 2>&1; c=$?; echo "[exit $c]" >> $LOG; return $c; }
want_fail() { if run "$@"; then echo "UNEXPECTED SUCCESS: $*" | tee -a $LOG; exit 1; fi; }

D=$(grep '^collection ' $LOG | tail -1 | cut -d' ' -f2); M=$(grep '^master ' $LOG | tail -1 | cut -d' ' -f2)
echo "D=$D M=$M" | tee -a $LOG
# the new definition and master accounts pay their own creation? No: --payer A1 pays.
run nft info $D --json || exit 1
run nft init --definition $D --account $A2 || exit 1

run nft print --from $M --to Public/$A2 || exit 1                  # master 3 -> 2, A2 owns a copy
want_fail nft send --from $A2 --to Public/$A4                     # A4 not ready: refused before sending
run nft init --definition $D --account $A3 || exit 1
want_fail nft init --definition $D --new public                   # no payer for a new account: refused, no account made
run nft init --definition $D --new public --fee-payer $A1 || exit 1   # a new public holding, A1 pays (prints "new account Public/<id>")
run nft send --from $A2 --to Public/$A3 || exit 1                  # copy A2 -> A3
run nft list --json || exit 1
echo '{"action":"send","from":"Public/'$A3'","to":"Public/'$A2'"}' > $W/plan-in.json
$N module nft plan < $W/plan-in.json >> $LOG 2>&1; echo "[exit $?] (plan)" >> $LOG
run nft burn --from $A3 || exit 1                                  # the copy is gone; supply 3 -> 2
run nft list --json || exit 1
$N module nft holdings >> $LOG 2>&1; echo "[exit $?] (module holdings)" >> $LOG
echo "PUBLIC E2E DONE" | tee -a $LOG
E2E
